%% ====== SLIDE: WHAT THE GUARDRAIL TAUGHT ME ======
\begin{frame}{What ten weeks of running a guardrail taught me}
\vskip-4pt
{\small In June I showed a list of file types it blocks. Here's what actually happened after.\par}
\vskip2pt
\begin{columns}[T]
\begin{column}{0.48\textwidth}
\begin{tcolorbox}[dontbox, title={\faEyeSlash\quad It failed silently}, top=2pt, bottom=2pt]
\scriptsize A one-line shell quirk made it fail \emph{open} for about a month. Nothing told me. Fix: it now logs one line every time it runs --- and it still crashes now and then (17 times since June; twice this week).
\end{tcolorbox}
\vskip3pt
\begin{tcolorbox}[dontbox, title={\faBan\quad It over-blocks}, top=2pt, bottom=2pt]
\scriptsize It matches the \emph{text} of a command, not what the command does. This week it refused to write \textbf{this slide}, because the slide mentioned \texttt{.csv} and \texttt{grep}.
\end{tcolorbox}
\end{column}
\begin{column}{0.48\textwidth}
\begin{tcolorbox}[keybox, title={\faFolderOpen\quad The leak wasn't in the files}, top=2pt, bottom=2pt]
\scriptsize Brightspace puts student names in the \emph{folder titles}. Listing the folder was the disclosure. The guard now blocks \emph{listing} those folders, not just reading them.
\end{tcolorbox}
\vskip3pt
\begin{tcolorbox}[keybox, title={\faHandPaper\quad Deliberate holes}, top=2pt, bottom=2pt]
\scriptsize Spreadsheets aren't blocked (too much friction for tables I make myself); one mail script is allow-listed. Every hole is a judgment call, written down in the script.
\end{tcolorbox}
\end{column}
\end{columns}
\vskip2pt
\begin{tcolorbox}[quotebox, top=1pt, bottom=1pt]
\scriptsize \textbf{Since June:} $\sim$62,000 tool calls $\cdot$ $\sim$640 blocked ($\sim$1\%) $\cdot$ 17 silent crashes. \textbf{The principle survived} --- a script may read PII, the agent may not. The implementation is disposable, and I expect to rewrite it.
\end{tcolorbox}
\end{frame}

\begin{frame}{What the agent sees --- and what stays yours}

\begin{columns}[T]

\begin{tcolorbox}[pipelinebox, title={\faEye\quad The agent receives}]
\scriptsize
\texttt{anon\_id: STU04}\\
\texttt{Question 1: Describe the economic tradeoff\ldots}\\
\texttt{Student response: The opportunity cost\ldots [391 words]}\\[3pt]
No name. No student ID. No course section.
\end{tcolorbox}
\vskip3pt
\begin{tcolorbox}[pipelinebox, title={\faBrain\quad The agent returns}]
\scriptsize
\begin{itemize}\setlength{\itemsep}{1pt}
  \item \textbf{Score} (0 / 1 / 2 per rubric)
  \item \textbf{Justification} (1--2 sentences)
  \item \textbf{Review queue}: flag if borderline
\end{itemize}
\end{tcolorbox}
\end{column}
\begin{column}{0.48\textwidth}
\begin{tcolorbox}[keybox, title={\faUserCheck\quad Your judgment}]
\scriptsize
\begin{itemize}\setlength{\itemsep}{2pt}
  \item \textbf{Edge cases}: embedded links, borderline word counts
  \item \textbf{Rubric calls}: you set the rubric, review anything the explanation doesn't support
  \item \textbf{Final sign-off}: the grade sheet is yours before upload
\end{itemize}
\end{tcolorbox}
\vskip3pt
\begin{tcolorbox}[accentbox, top=3pt, bottom=3pt]
\scriptsize
The pipeline handles \textbf{mechanical work at scale}. \textbf{Pedagogical judgment} --- what counts as thoughtful, what warrants a flag --- is \textbf{still yours}.
\end{tcolorbox}
\end{column}
\end{columns}
\end{frame}
